```python
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import pickle
import os

def load_and_prepare_data(data_path='../../data/creditcard.csv', test_size=0.2, random_state=42):
    """
    Load and prepare the fraud detection dataset
    """
    print("Loading data...")
    df = pd.read_csv(data_path)

    print(f"Dataset shape: {df.shape}")
    print(f"Fraud cases: {df['Class'].sum()} ({df['Class'].mean()*100:.2f}%)")

    # Separate features and target
    X = df.drop('Class', axis=1)
    y = df['Class']

    # Split the data
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=test_size, random_state=random_state, stratify=y
    )

    print(f"\nTrain set shape: {X_train.shape}")
    print(f"Test set shape: {X_test.shape}")
    print(f"Train fraud rate: {y_train.mean()*100:.2f}%")
    print(f"Test fraud rate: {y_test.mean()*100:.2f}%")

    # Scale features (excluding Time column if you want)
    scaler = StandardScaler()

    # Fit on training data only
    X_train_scaled = scaler.fit_transform(X_train)
    X_test_scaled = scaler.transform(X_test)

    # Convert back to DataFrame to preserve column names
    X_train_scaled = pd.DataFrame(X_train_scaled, columns=X_train.columns, index=X_train.index)
    X_test_scaled = pd.DataFrame(X_test_scaled, columns=X_test.columns, index=X_test.index)

    # Save processed data
    processed_dir = '../../data/processed'
    os.makedirs(processed_dir, exist_ok=True)

    X_train_scaled.to_csv(f'{processed_dir}/X_train.csv', index=False)
    X_test_scaled.to_csv(f'{processed_dir}/X_test.csv', index=False)
    y_train.to_csv(f'{processed_dir}/y_train.csv', index=False)
    y_test.to_csv(f'{processed_dir}/y_test.csv', index=False)

    # Save scaler for later use in production
    with open(f'{processed_dir}/scaler.pkl', 'wb') as f:
        pickle.dump(scaler, f)

    print(f"\nProcessed data saved to {processed_dir}/")

    return X_train_scaled, X_test_scaled, y_train, y_test, scaler

if __name__ == "__main__":
    X_train, X_test, y_train, y_test, scaler = load_and_prepare_data()
```

```python
import mlflow
import mlflow.sklearn
import pandas as pd
import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import (
    roc_auc_score, precision_score, recall_score,
    f1_score, confusion_matrix, classification_report,
    precision_recall_curve, roc_curve
)
import matplotlib.pyplot as plt
import seaborn as sns
import warnings
warnings.filterwarnings('ignore')

def load_processed_data():
    """Load the processed training and test data"""
    processed_dir = '../../data/processed'

    X_train = pd.read_csv(f'{processed_dir}/X_train.csv')
    X_test = pd.read_csv(f'{processed_dir}/X_test.csv')
    y_train = pd.read_csv(f'{processed_dir}/y_train.csv').values.ravel()
    y_test = pd.read_csv(f'{processed_dir}/y_test.csv').values.ravel()

    return X_train, X_test, y_train, y_test

def plot_confusion_matrix(y_true, y_pred, save_path='confusion_matrix.png'):
    """Plot and save confusion matrix"""
    cm = confusion_matrix(y_true, y_pred)

    plt.figure(figsize=(8, 6))
    sns.heatmap(cm, annot=True, fmt='d', cmap='Blues')
    plt.title('Confusion Matrix')
    plt.ylabel('True Label')
    plt.xlabel('Predicted Label')
    plt.tight_layout()
    plt.savefig(save_path)
    plt.close()

    return save_path

def plot_roc_curve(y_true, y_pred_proba, save_path='roc_curve.png'):
    """Plot and save ROC curve"""
    fpr, tpr, _ = roc_curve(y_true, y_pred_proba)
    auc = roc_auc_score(y_true, y_pred_proba)

    plt.figure(figsize=(8, 6))
    plt.plot(fpr, tpr, label=f'ROC curve (AUC = {auc:.3f})')
    plt.plot([0, 1], [0, 1], 'k--', label='Random')
    plt.xlabel('False Positive Rate')
    plt.ylabel('True Positive Rate')
    plt.title('ROC Curve')
    plt.legend()
    plt.grid(True)
    plt.tight_layout()
    plt.savefig(save_path)
    plt.close()

    return save_path

def plot_precision_recall_curve(y_true, y_pred_proba, save_path='pr_curve.png'):
    """Plot and save Precision-Recall curve"""
    precision, recall, _ = precision_recall_curve(y_true, y_pred_proba)

    plt.figure(figsize=(8, 6))
    plt.plot(recall, precision)
    plt.xlabel('Recall')
    plt.ylabel('Precision')
    plt.title('Precision-Recall Curve')
    plt.grid(True)
    plt.tight_layout()
    plt.savefig(save_path)
    plt.close()

    return save_path

def train_logistic_regression(X_train, X_test, y_train, y_test, class_weight='balanced'):
    """Train Logistic Regression with MLflow tracking"""

    # Set experiment
    mlflow.set_experiment("fraud-detection-baseline")

    with mlflow.start_run(run_name="logistic_regression_baseline"):

        # Log parameters
        mlflow.log_param("model_type", "logistic_regression")
        mlflow.log_param("class_weight", class_weight)
        mlflow.log_param("max_iter", 1000)
        mlflow.log_param("solver", "lbfgs")

        # Train model
        print("Training Logistic Regression...")
        model = LogisticRegression(
            class_weight=class_weight,
            max_iter=1000,
            solver='lbfgs',
            random_state=42
        )
        model.fit(X_train, y_train)

        # Predictions
        y_pred = model.predict(X_test)
        y_pred_proba = model.predict_proba(X_test)[:, 1]

        # Calculate metrics
        auc = roc_auc_score(y_test, y_pred_proba)
        precision = precision_score(y_test, y_pred)
        recall = recall_score(y_test, y_pred)
        f1 = f1_score(y_test, y_pred)

        # Log metrics
        mlflow.log_metric("auc_roc", auc)
        mlflow.log_metric("precision", precision)
        mlflow.log_metric("recall", recall)
        mlflow.log_metric("f1_score", f1)

        print(f"\nLogistic Regression Results:")
        print(f"AUC-ROC: {auc:.4f}")
        print(f"Precision: {precision:.4f}")
        print(f"Recall: {recall:.4f}")
        print(f"F1-Score: {f1:.4f}")

        # Create and log plots
        cm_path = plot_confusion_matrix(y_test, y_pred, 'lr_confusion_matrix.png')
        roc_path = plot_roc_curve(y_test, y_pred_proba, 'lr_roc_curve.png')
        pr_path = plot_precision_recall_curve(y_test, y_pred_proba, 'lr_pr_curve.png')

        mlflow.log_artifact(cm_path)
        mlflow.log_artifact(roc_path)
        mlflow.log_artifact(pr_path)

        # Log model
        mlflow.sklearn.log_model(model, "model")

        # Print classification report
        print("\nClassification Report:")
        print(classification_report(y_test, y_pred))

        return model, auc

if __name__ == "__main__":
    # Set MLflow tracking URI
    mlflow.set_tracking_uri("http://localhost:5000")

    # Load data
    print("Loading processed data...")
    X_train, X_test, y_train, y_test = load_processed_data()
```

```python
import mlflow
import mlflow.sklearn
import mlflow.xgboost
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from xgboost import XGBClassifier
from lightgbm import LGBMClassifier
from sklearn.metrics import roc_auc_score, precision_score, recall_score, f1_score
from imblearn.over_sampling import SMOTE
from imblearn.under_sampling import RandomUnderSampler
from imblearn.pipeline import Pipeline as ImbPipeline
import warnings
warnings.filterwarnings('ignore')

def load_processed_data():
    """Load the processed training and test data"""
    processed_dir = '../../data/processed'

    X_train = pd.read_csv(f'{processed_dir}/X_train.csv')
    X_test = pd.read_csv(f'{processed_dir}/X_test.csv')
    y_train = pd.read_csv(f'{processed_dir}/y_train.csv').values.ravel()
    y_test = pd.read_csv(f'{processed_dir}/y_test.csv').values.ravel()

    return X_train, X_test, y_train, y_test

def train_with_sampling(X_train, X_test, y_train, y_test, sampling_strategy='SMOTE'):
    """Train model with resampling techniques"""

    mlflow.set_experiment("fraud-detection-sampling")

    with mlflow.start_run(run_name=f"xgboost_{sampling_strategy}"):

        # Log parameters
        mlflow.log_param("model_type", "xgboost")
        mlflow.log_param("sampling_strategy", sampling_strategy)

        # Apply sampling
        if sampling_strategy == 'SMOTE':
            sampler = SMOTE(random_state=42)
            X_train_resampled, y_train_resampled = sampler.fit_resample(X_train, y_train)
        elif sampling_strategy == 'undersample':
            sampler = RandomUnderSampler(random_state=42)
            X_train_resampled, y_train_resampled = sampler.fit_resample(X_train, y_train)
        else:
            X_train_resampled, y_train_resampled = X_train, y_train

        print(f"\nTraining with {sampling_strategy}...")
        print(f"Original training size: {len(y_train)}")
        print(f"Resampled training size: {len(y_train_resampled)}")
        print(f"Resampled fraud rate: {y_train_resampled.mean()*100:.2f}%")

        # Train XGBoost
        model = XGBClassifier(
            n_estimators=100,
            max_depth=6,
            learning_rate=0.1,
            subsample=0.8,
            colsample_bytree=0.8,
            random_state=42,
            eval_metric='logloss'
        )

        mlflow.log_param("n_estimators", 100)
        mlflow.log_param("max_depth", 6)
        mlflow.log_param("learning_rate", 0.1)

        model.fit(X_train_resampled, y_train_resampled)

        # Predictions
        y_pred_proba = model.predict_proba(X_test)[:, 1]
        y_pred = model.predict(X_test)

        # Calculate metrics
        auc = roc_auc_score(y_test, y_pred_proba)
        precision = precision_score(y_test, y_pred)
        recall = recall_score(y_test, y_pred)
        f1 = f1_score(y_test, y_pred)

        # Log metrics
        mlflow.log_metric("auc_roc", auc)
        mlflow.log_metric("precision", precision)
        mlflow.log_metric("recall", recall)
        mlflow.log_metric("f1_score", f1)

        print(f"\nResults with {sampling_strategy}:")
        print(f"AUC-ROC: {auc:.4f}")
        print(f"Precision: {precision:.4f}")
        print(f"Recall: {recall:.4f}")
        print(f"F1-Score: {f1:.4f}")

        # Log model
        mlflow.xgboost.log_model(model, "model")

        return model, auc

def train_lightgbm(X_train, X_test, y_train, y_test):
    """Train LightGBM model"""

    mlflow.set_experiment("fraud-detection-advanced")

    with mlflow.start_run(run_name="lightgbm_baseline"):

        # Calculate scale_pos_weight for imbalanced data
        scale_pos_weight = (len(y_train) - y_train.sum()) / y_train.sum()

        mlflow.log_param("model_type", "lightgbm")
        mlflow.log_param("scale_pos_weight", scale_pos_weight)

        print("\nTraining LightGBM...")

        model = LGBMClassifier(
            n_estimators=100,
            max_depth=6,
            learning_rate=0.1,
            num_leaves=31,
            scale_pos_weight=scale_pos_weight,
            random_state=42,
            verbose=-1
        )

        mlflow.log_param("n_estimators", 100)
        mlflow.log_param("max_depth", 6)
        mlflow.log_param("learning_rate", 0.1)

        model.fit(X_train, y_train)

        # Predictions
        y_pred_proba = model.predict_proba(X_test)[:, 1]
        y_pred = model.predict(X_test)

        # Calculate metrics
        auc = roc_auc_score(y_test, y_pred_proba)
        precision = precision_score(y_test, y_pred)
        recall = recall_score(y_test, y_pred)
        f1 = f1_score(y_test, y_pred)

        # Log metrics
        mlflow.log_metric("auc_roc", auc)
        mlflow.log_metric("precision", precision)
        mlflow.log_metric("recall", recall)
        mlflow.log_metric("f1_score", f1)

        print(f"\nLightGBM Results:")
        print(f"AUC-ROC: {auc:.4f}")
        print(f"Precision: {precision:.4f}")
        print(f"Recall: {recall:.4f}")
        print(f"F1-Score: {f1:.4f}")

        # Log model
        mlflow.lightgbm.log_model(model, "model")

        return model, auc

if __name__ == "__main__":
    # Set MLflow tracking URI
    mlflow.set_tracking_uri("http://localhost:5000")

    # Load data
    print("Loading processed data...")
    X_train, X_test, y_train, y_test = load_processed_data()

    # Train multiple models
    models_performance = {}

    # 1. XGBoost with SMOTE
    model_smote, auc_smote = train_with_sampling(X_train, X_test, y_train, y_test, 'SMOTE')
    models_performance['XGBoost_SMOTE'] = auc_smote

    # 2. XGBoost with undersampling
```

```python
import mlflow
from mlflow.tracking import MlflowClient

def register_best_model(experiment_name, model_name="fraud-detection-model"):
    """
    Find the best model from an experiment and register it
    """
    client = MlflowClient()

    # Get experiment
    experiment = client.get_experiment_by_name(experiment_name)
    if experiment is None:
        print(f"Experiment '{experiment_name}' not found!")
        return

    # Get all runs from the experiment
    runs = client.search_runs(
        experiment_ids=[experiment.experiment_id],
        order_by=["metrics.auc_roc DESC"],
        max_results=1
    )

    if not runs:
        print("No runs found in the experiment!")
        return

    best_run = runs[0]
    best_auc = best_run.data.metrics.get('auc_roc', 0)

    print(f"\n🔍 Best run found:")
    print(f"   Run ID: {best_run.info.run_id}")
    print(f"   AUC-ROC: {best_auc:.4f}")
    print(f"   Model: {best_run.data.params.get('model_type', 'unknown')}")

    # Register the model
    model_uri = f"runs:/{best_run.info.run_id}/model"

    try:
        registered_model = mlflow.register_model(model_uri, model_name)
        print(f"\n✅ Model registered successfully!")
        print(f"   Name: {model_name}")
        print(f"   Version: {registered_model.version}")

        # Add description and tags
        client.update_model_version(
            name=model_name,
            version=registered_model.version,
            description=f"Fraud detection model with AUC-ROC: {best_auc:.4f}"
        )

        # Transition to staging
        client.transition_model_version_stage(
            name=model_name,
            version=registered_model.version,
            stage="Staging"
        )
        print(f"   Stage: Staging")

    except Exception as e:
        print(f"Error registering model: {e}")

if __name__ == "__main__":
    mlflow.set_tracking_uri("http://localhost:5000")

    # Register best model from advanced experiments
    register_best_model("fraud-detection-advanced", "fraud-detection-model")

    print("\n📊 View your registered model at:")
    print("   http://localhost:5000/#/models/fraud-detection-model")
```
